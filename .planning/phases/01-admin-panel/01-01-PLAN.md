<name>Task 2: Create auth route and JWT middleware</name>
<files>
  server/routes/auth.ts

<action>
1. Create `server/middleware/auth.ts`:
   ```typescript
   import { Request, Response, NextFunction } from 'express';
   import jwt from 'jsonwebtoken';

   const JWT_SECRET = 'impact-graphics-admin-secret-2024';

   export interface AuthRequest extends Request {
     admin?: { email: string };
   }

   export function generateToken(email: string): string {
     return jwt.sign({ email }, JWT_SECRET, { expiresIn: '24h' });
   }

   export function authenticateToken(req: AuthRequest, res: Response, next: NextFunction): void {
     const authHeader = req.headers['authorization'];
     const token = authHeader && authHeader.split(' ')[1]; // Bearer TOKEN

     if (!token) {
       res.status(401).json({ error: 'Access denied. No token provided.' });
       return;
     }

     try {
       const decoded = jwt.verify(token, JWT_SECRET) as { email: string };
       req.admin = decoded;
       next();
     } catch {
       res.status(403).json({ error: 'Invalid or expired token.' });
     }
   }
   ```

2. Create `server/routes/auth.ts`:
   ```typescript
   import { Router, Request, Response } from 'express';
   import { generateToken } from '../middleware/auth';

   const router = Router();

   const ADMIN_EMAIL = 'kajal00@gmail.com';
   const ADMIN_PASSWORD = '11223344';

   router.post('/login', (req: Request, res: Response) => {
     const { email, password } = req.body;

     if (!email || !password) {
       res.status(400).json({ error: 'Email and password are required.' });
       return;
     }

     if (email !== ADMIN_EMAIL || password !== ADMIN_PASSWORD) {
       res.status(401).json({ error: 'Invalid email or password.' });
       return;
     }

     const token = generateToken(email);
     res.json({ token, email });
   });

   export default router;
   ```
</action>

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → API | Public internet-facing API endpoints |
| Unauthenticated → Protected Routes | Bookings GET requires valid JWT |
| Booking Form → Server | Public form submission from main site |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-admin-01 | Spoofing | POST /api/auth/login | mitigate | Validate credentials server-side, never expose in client code |
| T-admin-02 | Tampering | POST /api/bookings | mitigate | Validate required fields (name, phone, email, service, description) on server; reject malformed data |
| T-admin-03 | Information Disclosure | GET /api/bookings | mitigate | authenticateToken middleware blocks unauthenticated requests; 401 with no data leaked |
| T-admin-04 | Information Disclosure | bookings.json | mitigate | File is in server/ directory, not in public/ — never served by Vite |
| T-admin-05 | Repudiation | Booking creation | accept | No audit trail beyond createdAt timestamp; acceptable for v1 |
</threat_model>
